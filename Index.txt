// --- DATE UTILITIES ---
const formatDateKey = (date) => {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, '0');
  const d = String(date.getDate()).padStart(2, '0');
  return `${y}-${m}-${d}`;
};

const addDays = (date, days) => {
  const copy = new Date(date);
  copy.setDate(copy.getDate() + days);
  return copy;
};

const isSameDay = (d1, d2) => formatDateKey(d1) === formatDateKey(d2);

const isPastDate = (targetDate, todayDate) => {
  const tKey = formatDateKey(targetDate);
  const todayKey = formatDateKey(todayDate);
  return tKey < todayKey;
};

const isFutureDate = (targetDate, todayDate) => {
  const tKey = formatDateKey(targetDate);
  const todayKey = formatDateKey(todayDate);
  return tKey > todayKey;
};

// Generate realistic 30-day history so "How's My Routine" & Locked Past Days are immediately rich and testable
const generateSeededHistory = (todayDate, tasks) => {
  const history = {};
  for (let i = 1; i <= 28; i++) {
    const pastD = addDays(todayDate, -i);
    const key = formatDateKey(pastD);
    const dayLog = {};
    // Deterministic pseudo-random based on day index so it stays consistent
    tasks.forEach((task, idx) => {
      const seed = (i * 7 + idx * 13) % 10;
      // Prayers have high completion (~88%), other habits ~78%
      const threshold = task.category === 'prayers' ? 1 : 2;
      dayLog[task.id] = {
        completed: seed >= threshold,
        completedAt: `${task.time}`,
        note: task.id === 'sleeping' ? `${7 + (seed % 2)}.5 hrs restful sleep` : ''
      };
    });
    history[key] = dayLog;
  }
  // Pre-check a couple of morning items for today
  const todayKey = formatDateKey(todayDate);
  history[todayKey] = {
    fajr: { completed: true, completedAt: '05:12' },
    facial_am: { completed: true, completedAt: '07:35' },
    medicine_am: { completed: true, completedAt: '08:32' }
  };
  return history;
};

// --- APPLE CALENDAR (.ICS) GENERATOR ---
const generateICSFile = (tasksToExport, dateObj, singleTask = null) => {
  const items = singleTask ? [singleTask] : tasksToExport;
  const y = dateObj.getFullYear();
  const m = String(dateObj.getMonth() + 1).padStart(2, '0');
  const d = String(dateObj.getDate()).padStart(2, '0');

  let icsEvents = '';
  items.forEach((task) => {
    const [hh, mm] = (task.time || '09:00').split(':');
    const startStr = `${y}${m}${d}T${hh}${mm}00`;
    const endHour = String((parseInt(hh, 10) + 1) % 24).padStart(2, '0');
    const endStr = `${y}${m}${d}T${endHour}${mm}00`;
    const rrule = task.isDaily ? '\nRRULE:FREQ=DAILY' : '';

    icsEvents += `BEGIN:VEVENT
UID:firdaus-${task.id}-${y}${m}${d}@firdausreminder.mv
DTSTAMP:${y}${m}${d}T000000Z
DTSTART:${startStr}
DTEND:${endStr}${rrule}
SUMMARY:🌸 ${task.title} (Firdaus Reminder)
DESCRIPTION:${task.subtitle || 'Daily routine reminder from Firdaus Reminder'}
BEGIN:VALARM
TRIGGER:-PT5M
ACTION:DISPLAY
DESCRIPTION:Reminder: ${task.title}
END:VALARM
END:VEVENT\n`;
  });

  const icsContent = `BEGIN:VCALENDAR
VERSION:2.0
PRODID:-//Firdaus Reminder//Maldives Routine Tracker//EN
CALSCALE:GREGORIAN
METHOD:PUBLISH
X-WR-CALNAME:Firdaus Reminder
${icsEvents}END:VCALENDAR`;

  const blob = new Blob([icsContent], { type: 'text/calendar;charset=utf-8' });
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  const fileName = singleTask
    ? `firdaus-${singleTask.id}.ics`
    : `firdaus-routine-${y}-${m}-${d}.ics`;
  link.setAttribute('download', fileName);
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
};

          }

      return {
        date: d,
        dateKey: dKey,
        dayLabel: d.toLocaleDateString('en-US', { weekday: 'short' }),
        shortDate: `${d.getDate()}/${d.getMonth() + 1}`,

      prayerRate,
      totalCompleted,
      totalPossible,
      streak,
      dailyBreakdown,
      categoryTotals

                <div className="flex flex-wrap items-center justify-between gap-3 mb-3">

                  </div>

                  <div className="flex items-center gap-2">
                    <button
                      onClick={() => {
                        generateICSFile(tasksForSelectedDate, selectedDate);

                          'Synced to Apple Calendar 🍎',
                          'Downloaded .ics schedule with push alarms for your iPhone/iPad Calendar.',

                      }}
                      style={{ backgroundColor: palette.card, borderColor: palette.border }}
                      className="px-3 py-2 rounded-xl border text-xs font-semibold flex items-center gap-1.5 shadow-sm hover:opacity-90"
                    >
                      <Download className="w-3.5 h-3.5" />
                      Sync to Apple Calendar
                    </button>

                    <button
                      onClick={() => setShowAddTaskModal(true)}
                      style={{ backgroundColor: palette.buttonBg, color: palette.buttonText }}
                      className="px-3.5 py-2 rounded-xl text-xs font-bold flex items-center gap-1.5 shadow-sm hover:opacity-95"
                    >
                      <Plus className="w-4 h-4" />
                      Add Task
                    </button>
                  </div>
                </div>

                {/* Progress Bar */}
                <div
                  style={{ backgroundColor: palette.progressTrack }}
                  className="w-full h-3 rounded-full overflow-hidden"
                >
                  <div
                    style={{
                      width: `${selectedDayProgress.pct}%`,
                      backgroundColor: palette.accent
                    }}
                    className="h-full rounded-full transition-all duration-500"
                  />
                </div>
              </div>
            )}

            {/* CATEGORY PILLS FILTER */}
            <div className="flex items-center gap-2 overflow-x-auto pb-1 no-scrollbar">
              <button
                onClick={() => setSelectedCategoryFilter('all')}
                style={
                  selectedCategoryFilter === 'all'
                    ? { backgroundColor: palette.buttonBg, color: palette.buttonText }
                    : { backgroundColor: palette.card, color: palette.textMuted, borderColor: palette.border }
                }
                className="px-3.5 py-1.5 rounded-full text-xs font-semibold border shrink-0 transition"
              >
                All Daily Stuffs ({tasks.filter((t) => t.isDaily || t.specificDate === selectedDateKey).length})
              </button>
              {Object.entries(CATEGORIES).map(([catKey, catInfo]) => {
                const IconComp = catInfo.icon;
                const isSel = selectedCategoryFilter === catKey;
                return (
                  <button
                    key={catKey}
                    onClick={() => setSelectedCategoryFilter(catKey)}
                    style={
                      isSel
                        ? { backgroundColor: palette.buttonBg, color: palette.buttonText }
                        : { backgroundColor: palette.card, color: palette.textMuted, borderColor: palette.border }
                    }
                    className="px-3.5 py-1.5 rounded-full text-xs font-semibold border shrink-0 flex items-center gap-1.5 transition"
                  >
                    <IconComp className="w-3.5 h-3.5" />
                    {catInfo.label}
                  </button>
                );
              })}
            </div>

            {/* DAILY CHECKLIST CARDS */}
            <div className="space-y-2.5">
              {tasksForSelectedDate.map((task) => {
                const dayRecord = logs[selectedDateKey] || {};
                const isCompleted = !!dayRecord[task.id]?.completed;
                const completedAt = dayRecord[task.id]?.completedAt;
                const catObj = CATEGORIES[task.category] || CATEGORIES.custom;
                const CatIcon = catObj.icon;

                return (
                  <div
                    key={task.id}
                    style={{
                      backgroundColor: isCompleted ? palette.cardAlt : palette.card,
                      borderColor: isCompleted ? palette.accent : palette.border
                    }}
                    className={`rounded-2xl p-3.5 sm:p-4 border transition flex items-center justify-between gap-3 ${
                      isSelectedDayPast ? 'opacity-85' : 'hover:shadow-sm'
                    }`}
                  >
                    <div
                      onClick={() => toggleTaskCompletion(task)}
                      className="flex items-start gap-3.5 flex-1 cursor-pointer select-none"
                    >
                      {/* Checkbox or Lock Icon */}
                      <button
                        type="button"
                        disabled={isSelectedDayPast}
                        className="mt-0.5 shrink-0 focus:outline-none"
                      >
                        {isCompleted ? (
                          <CheckCircle2
                            className="w-6 h-6"
                            style={{ color: palette.accent }}
                          />
                        ) : isSelectedDayPast ? (
                          <Lock className="w-5 h-5 opacity-50" />
                        ) : (
                          <Circle
                            className="w-6 h-6 opacity-60"
                            style={{ color: palette.textMuted }}
                          />

                      </button>

                      <div className="flex-1 min-w-0">
                        <div className="flex flex-wrap items-center gap-2">
                          <h4
                            className={`font-bold text-sm sm:text-base ${
                              isCompleted ? 'line-through opacity-75' : ''
                            }`}
                          >
                            {task.title}
                          </h4>
                          <span
                            style={{ backgroundColor: palette.bg, borderColor: palette.border, color: palette.textMuted }}
                            className="text-[10px] px-2 py-0.5 rounded-full border font-medium flex items-center gap-1"
                          >
                            <CatIcon className="w-3 h-3" />
                            {catObj.label}
                          </span>
                        </div>

                        <p className="text-xs mt-0.5" style={{ color: palette.textMuted }}>
                          {task.subtitle}
                        </p>

                        {/* Special Sleep Hours Adjuster when task is 'sleeping' */}
                        {task.id === 'sleeping' && (
                          <div
                            onClick={(e) => e.stopPropagation()}
                            className="mt-2 inline-flex items-center gap-2 text-xs px-2.5 py-1 rounded-lg border"
                            style={{ backgroundColor: palette.bg, borderColor: palette.border }}
                          >
                            <BedDouble className="w-3.5 h-3.5" style={{ color: palette.accent }} />
                            <span>Sleep duration:</span>
                            <input
                              type="number"
                              min={3}
                              max={12}
                              step={0.5}
                              disabled={isSelectedDayPast}
                              value={sleepHoursLog[selectedDateKey] ?? 8}
                              onChange={(e) =>
                                setSleepHoursLog({
                                  ...sleepHoursLog,
                                  [selectedDateKey]: parseFloat(e.target.value) || 8
                                })
                              }
                              style={{ backgroundColor: palette.card, borderColor: palette.border }}
                              className="w-14 px-1.5 py-0.5 rounded border text-center font-bold"
                            />
                            <span>hrs</span>
                          </div>
                        )}

                        <div className="flex items-center gap-3 mt-2 text-[11px]" style={{ color: palette.textMuted }}>
                          <span className="flex items-center gap-1 font-medium">
                            <Clock className="w-3 h-3" />
                            {task.time}
                          </span>
                          {isCompleted && completedAt && (
                            <span className="font-semibold" style={{ color: palette.accent }}>
                              ✓ Marked done at {completedAt}
                            </span>
                          )}
                          {isSelectedDayPast && !isCompleted && (
                            <span className="opacity-75">Missed (Locked)</span>
                          )}
                        </div>
                      </div>
                    </div>

                    {/* Right Actions: Push Notification & Apple Calendar .ics Alert */}
                    <div className="flex items-center gap-1.5 shrink-0">
                      <button
                        type="button"
                        onClick={() => handleTestPushNotification(task)}
                        style={{ backgroundColor: palette.bg, borderColor: palette.border }}
                        className="p-2 rounded-xl border hover:opacity-80 transition"
                        title="Send Push Notification Reminder"
                      >
                        <Bell className="w-4 h-4" style={{ color: palette.accent }} />
                      </button>

                      <button
                        type="button"
                        onClick={() => {
                          generateICSFile(tasks, selectedDate, task);
                          triggerToast(
                            'Apple Calendar Reminder Ready 🍎',
                            `Added "${task.title}" (${task.time}) push notification to Apple Calendar.`,
                            'success'
                          );
                        }}
                        style={{ backgroundColor: palette.bg, borderColor: palette.border }}
                        className="px-2.5 py-2 rounded-xl border text-[11px] font-semibold flex items-center gap-1 hover:opacity-80 transition"
                        title="Add to Apple Calendar with Push Alarm"
                      >
                        <span></span>
                        <span className="hidden sm:inline">Alert</span>
                      </button>

                      {task.id.startsWith('custom_') && !isSelectedDayPast && (
                        <button
                          type="button"
                          onClick={() => handleDeleteCustomTask(task.id)}
                          style={{ backgroundColor: palette.bg, borderColor: palette.border }}
                          className="p-2 rounded-xl border text-red-500 hover:opacity-80 transition"
                          title="Delete Custom Task"
                        >
                          <Trash2 className="w-4 h-4" />
                        </button>
                      )}
                    </div>
                  </div>
                );
              })}
            </div>
          </>
        )}

        {/* ====================================================================
            TAB 2: "HOW'S MY ROUTINE" ANALYTICS TAB
           ==================================================================== */}
        {activeTab === 'analytics' && (
          <div className="space-y-5">
            {/* Header & Weekly / Monthly Toggle */}
            <div
              style={{ backgroundColor: palette.card, borderColor: palette.border }}
              className="rounded-3xl p-5 border shadow-sm flex flex-wrap items-center justify-between gap-4"
            >
              <div>
                <span className="text-xs font-bold uppercase tracking-wider" style={{ color: palette.accent }}>
                  Personal Reflection & Consistency
                </span>
                <h2 className="text-xl sm:text-2xl font-bold mt-0.5">How's My Routine</h2>
                <p className="text-xs mt-0.5" style={{ color: palette.textMuted }}>
                  Based on your locked past days & today's completed habits.
                </p>
              </div>

              <div
                style={{ backgroundColor: palette.bg, borderColor: palette.border }}
                className="flex p-1 rounded-2xl border"
              >
                <button
                  onClick={() => setAnalyticsPeriod('weekly')}
                  style={
                    analyticsPeriod === 'weekly'
                      ? { backgroundColor: palette.buttonBg, color: palette.buttonText }
                      : { color: palette.textMuted }
                  }
                  className="px-4 py-2 rounded-xl text-xs font-bold transition"
                >
                  Weekly (7 Days)
                </button>
                <button
                  onClick={() => setAnalyticsPeriod('monthly')}
                  style={
                    analyticsPeriod === 'monthly'
                      ? { backgroundColor: palette.buttonBg, color: palette.buttonText }
                      : { color: palette.textMuted }
                  }
                  className="px-4 py-2 rounded-xl text-xs font-bold transition"
                >
                  Monthly (30 Days)
                </button>
              </div>
            </div>

            {/* Key Summary Stat Cards */}
            <div className="grid grid-cols-1 sm:grid-cols-3 gap-3.5">
              <div
                style={{ backgroundColor: palette.cardAlt, borderColor: palette.border }}
                className="rounded-3xl p-4 border"
              >
                <div className="flex items-center justify-between">
                  <span className="text-xs font-semibold" style={{ color: palette.textMuted }}>
                    Overall Routine Score
                  </span>
                  <TrendingUp className="w-4 h-4" style={{ color: palette.accent }} />
                </div>
                <p className="text-3xl font-extrabold mt-2">{analyticsData.overallRate}%</p>
                <p className="text-xs mt-1" style={{ color: palette.textMuted }}>
                  {analyticsData.totalCompleted} of {analyticsData.totalPossible} tasks completed
                </p>
              </div>

              <div
                style={{ backgroundColor: palette.cardAlt, borderColor: palette.border }}
                className="rounded-3xl p-4 border"
              >
                <div className="flex items-center justify-between">
                  <span className="text-xs font-semibold" style={{ color: palette.textMuted }}>
                    Salah (Prayer) Consistency
                  </span>
                  <Sparkles className="w-4 h-4" style={{ color: palette.accent }} />
                </div>
                <p className="text-3xl font-extrabold mt-2">{analyticsData.prayerRate}%</p>
                <p className="text-xs mt-1" style={{ color: palette.textMuted }}>
                  Masha'Allah, steady daily prayers
                </p>
              </div>

              <div
                style={{ backgroundColor: palette.cardAlt, borderColor: palette.border }}
                className="rounded-3xl p-4 border"
              >
                <div className="flex items-center justify-between">
                  <span className="text-xs font-semibold" style={{ color: palette.textMuted }}>
                    Consistent Days Streak
                  </span>
                  <Flame className="w-4 h-4" style={{ color: palette.accent }} />
                </div>
                <p className="text-3xl font-extrabold mt-2">{analyticsData.streak} Days</p>
                <p className="text-xs mt-1" style={{ color: palette.textMuted }}>
                  Days with 65%+ routine completed
                </p>
              </div>
            </div>

            {/* Visual Bar Chart of Daily Completion */}
            <div
              style={{ backgroundColor: palette.card, borderColor: palette.border }}
              className="rounded-3xl p-5 border shadow-sm"
            >
              <div className="flex items-center justify-between mb-4">
                <div>
                  <h3 className="font-bold text-base">
                    {analyticsPeriod === 'weekly' ? 'Last 7 Days Trend' : 'Last 30 Days Completion Heatmap'}
                  </h3>
                  <p className="text-xs" style={{ color: palette.textMuted }}>
                    Tap any bar to view that locked day's record
                  </p>
                </div>
                <span
                  style={{ backgroundColor: palette.accentSoft, color: palette.text }}
                  className="text-xs px-2.5 py-1 rounded-full font-semibold"
                >
                  Locked History Verified 🔒
                </span>
              </div>

              <div className="flex items-end gap-1.5 sm:gap-2 h-40 pt-4 overflow-x-auto pb-2">
                {analyticsData.dailyBreakdown.map((item) => (
                  <button
                    key={item.dateKey}
                    onClick={() => {
                      setSelectedDate(item.date);
                      setActiveTab('routine');
                    }}
                    className="flex-1 min-w-[22px] flex flex-col items-center group"
                    title={`${item.dateKey}: ${item.percentage}% (${item.completed}/${item.total})`}
                  >
                    <span className="text-[10px] font-bold mb-1 opacity-0 group-hover:opacity-100 transition">
                      {item.percentage}%
                    </span>
                    <div
                      style={{ backgroundColor: palette.progressTrack }}
                      className="w-full h-24 rounded-xl overflow-hidden flex items-end p-0.5"
                    >
                      <div
                        style={{
                          height: `${Math.max(item.percentage, 8)}%`,
                          backgroundColor: palette.accent
                        }}
                        className="w-full rounded-lg transition-all duration-500"
                      />
                    </div>
                    <span className="text-[10px] font-semibold mt-1.5" style={{ color: palette.textMuted }}>
                      {analyticsPeriod === 'weekly' ? item.dayLabel : item.shortDate}
                    </span>
                  </button>
                ))}
              </div>
            </div>

            {/* Category-by-Category Breakdown */}
            <div
              style={{ backgroundColor: palette.card, borderColor: palette.border }}
              className="rounded-3xl p-5 border shadow-sm space-y-4"
            >
              <h3 className="font-bold text-base">Routine Breakdown by Category</h3>
              <div className="space-y-3.5">
                {Object.entries(CATEGORIES).map(([catKey, catInfo]) => {
                  const stats = analyticsData.categoryTotals[catKey] || { completed: 0, total: 0 };
                  if (stats.total === 0) return null;
                  const pct = Math.round((stats.completed / stats.total) * 100);
                  const CatIcon = catInfo.icon;

                  return (
                    <div key={catKey} className="space-y-1.5">
                      <div className="flex items-center justify-between text-xs sm:text-sm">
                        <span className="font-semibold flex items-center gap-2">
                          <CatIcon className="w-4 h-4" style={{ color: palette.accent }} />
                          {catInfo.label}
                        </span>
                        <span className="font-bold">
                          {pct}%{' '}
                          <span className="font-normal text-xs" style={{ color: palette.textMuted }}>
                            ({stats.completed}/{stats.total})
                          </span>
                        </span>
                      </div>
                      <div
                        style={{ backgroundColor: palette.progressTrack }}
                        className="w-full h-2.5 rounded-full overflow-hidden"
                      >
                        <div
                          style={{ width: `${pct}%`, backgroundColor: palette.accent }}
                          className="h-full rounded-full transition-all duration-500"
                        />
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>

            {/* Loving Encouragement Card */}
            <div
              style={{ backgroundColor: palette.cardAlt, borderColor: palette.border }}
              className="rounded-3xl p-5 border flex items-start gap-3.5"
            >
              <Award className="w-7 h-7 shrink-0 mt-0.5" style={{ color: palette.accent }} />
              <div className="text-xs sm:text-sm space-y-1">
                <h4 className="font-bold">Gentle Routine Note for {accountProfile.name} 💛</h4>
                <p style={{ color: palette.textMuted }} className="leading-relaxed">
                  {analyticsData.overallRate >= 80
                    ? "Masha'Allah! You have been wonderfully consistent with your prayers, facial & hair care, medicine, and reading. Keep nurturing your soul and health!"
                    : "Every small step counts! Even on busy days, your prayers and self-care moments bring peace and barakah. Tomorrow is a fresh page."}
                </p>
              </div>
            </div>
          </div>
        )}

        {/* ====================================================================
            TAB 3: FULL MONTH CALENDAR & APPLE CALENDAR SYNC VIEW
           ==================================================================== */}
        {activeTab === 'calendar' && (
          <div className="space-y-5">
            <div
              style={{ backgroundColor: palette.card, borderColor: palette.border }}
              className="rounded-3xl p-5 border shadow-sm"
            >
              <div className="flex flex-wrap items-center justify-between gap-3 mb-4">
                <div>
                  <h2 className="text-lg sm:text-xl font-bold">
                    {selectedDate.toLocaleDateString('en-US', { month: 'long', year: 'numeric' })}
                  </h2>
                  <p className="text-xs" style={{ color: palette.textMuted }}>
                    Past days are locked (🔒) • Tap any date to view or schedule reminders
                  </p>
                </div>
                <button
                  onClick={() => {
                    generateICSFile(tasks, selectedDate);
                    triggerToast(
                      'Full Routine Exported to Apple Calendar 🍎',
                      'Open the downloaded .ics file on your iPhone or Mac to enable native Apple Calendar push notifications.',
                      'success'
                    );
                  }}
                  style={{ backgroundColor: palette.buttonBg, color: palette.buttonText }}
                  className="px-4 py-2.5 rounded-xl text-xs font-bold flex items-center gap-2 shadow-sm"
                >
                  <Download className="w-4 h-4" />
                  Export All to Apple Calendar (.ics)
                </button>
              </div>

              {/* 28-Day Visual Overview Grid */}
              <div className="grid grid-cols-7 gap-1.5 sm:gap-2 text-center">
                {['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'].map((d) => (
                  <div key={d} className="text-[11px] font-bold py-1" style={{ color: palette.textMuted }}>
                    {d}
                  </div>
                ))}
                {Array.from({ length: 28 }, (_, idx) => {
                  const d = addDays(today, idx - 20);
                  const dKey = formatDateKey(d);
                  const isPast = isPastDate(d, today);
                  const isTod = isSameDay(d, today);
                  const isSel = isSameDay(d, selectedDate);
                  const dayLog = logs[dKey] || {};
                  const dayTasks = tasks.filter((t) => t.isDaily || t.specificDate === dKey);
                  const doneCount = dayTasks.filter((t) => dayLog[t.id]?.completed).length;
                  const pct = dayTasks.length ? Math.round((doneCount / dayTasks.length) * 100) : 0;

                  return (
                    <button
                      key={dKey}
                      onClick={() => {
                        setSelectedDate(d);
                        setActiveTab('routine');
                      }}
                      style={
                        isSel
                          ? { backgroundColor: palette.buttonBg, color: palette.buttonText, borderColor: palette.accent }
                          : { backgroundColor: palette.bg, color: palette.text, borderColor: palette.border }
                      }
                      className={`p-2 rounded-2xl border flex flex-col items-center justify-center transition ${
                        isTod ? 'ring-2' : ''
                      }`}
                    >
                      <div className="flex items-center gap-0.5 text-xs font-bold">
                        <span>{d.getDate()}</span>
                        {isPast && <Lock className="w-2.5 h-2.5 opacity-60" />}
                      </div>
                      <span className="text-[10px] mt-1 opacity-80">
                        {isPast || isTod ? `${pct}%` : 'Plan'}
                      </span>
                    </button>
                  );
                })}
              </div>
            </div>
          </div>
        )}

        {/* ====================================================================
            TAB 4: THEMES & ACCOUNT SETTINGS
           ==================================================================== */}
        {activeTab === 'settings' && (
          <div className="space-y-5">
            {/* Theme Studio Card */}
            <div
              style={{ backgroundColor: palette.card, borderColor: palette.border }}
              className="rounded-3xl p-5 border shadow-sm space-y-4"
            >
              <div className="flex items-center justify-between">
                <div>
                  <h2 className="text-lg font-bold">Theme & Color Palettes</h2>
                  <p className="text-xs" style={{ color: palette.textMuted }}>
                    Switch between Light/Dark mode, calming dull palettes, or your special Fresh Lemon theme.
                  </p>
                </div>
                <button
                  onClick={() => {
                    const nextMode = colorMode === 'light' ? 'dark' : 'light';
                    setColorMode(nextMode);
                    persistToCloud({ colorMode: nextMode });
                  }}
                  style={{ backgroundColor: palette.buttonBg, color: palette.buttonText }}
                  className="px-4 py-2 rounded-xl text-xs font-bold flex items-center gap-2 shadow-sm"
                >
                  {colorMode === 'light' ? <Moon className="w-4 h-4" /> : <Sun className="w-4 h-4" />}
                  {colorMode === 'light' ? 'Switch to Dark Mode' : 'Switch to Light Mode'}
                </button>
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                {Object.values(THEMES).map((t) => {
                  const p = t[colorMode];
                  const isSel = themeKey === t.id;
                  return (
                    <button
                      key={t.id}
                      onClick={() => {
                        setThemeKey(t.id);
                        persistToCloud({ themeKey: t.id });
                      }}
                      style={{
                        backgroundColor: p.bg,
                        borderColor: isSel ? p.accent : p.border,
                        color: p.text
                      }}
                      className={`p-4 rounded-2xl border-2 text-left flex items-center justify-between transition ${
                        isSel ? 'shadow-md' : 'opacity-85 hover:opacity-100'
                      }`}
                    >
                      <div>
                        <div className="flex items-center gap-2">
                          <span className="font-bold text-sm">{t.name}</span>
                          {t.isSpecial && (
                            <span
                              style={{ backgroundColor: p.accent, color: p.accentText }}
                              className="text-[10px] px-2 py-0.5 rounded-full font-bold"
                            >
                              🍋 Special
                            </span>
                          )}
                        </div>
                        <p className="text-xs mt-0.5" style={{ color: p.textMuted }}>
                          {t.badge} • {colorMode === 'light' ? 'Light' : 'Dark'}
                        </p>
                      </div>
                      <div className="flex items-center gap-1.5">
                        <span
                          style={{ backgroundColor: p.bgSecondary, borderColor: p.border }}
                          className="w-5 h-5 rounded-full border"
                        />
                        <span
                          style={{ backgroundColor: p.accent }}
                          className="w-5 h-5 rounded-full"
                        />
                      </div>
                    </button>
                  );
                })}
              </div>
            </div>

            {/* Account & Maldives Registration Details */}
            <div
              style={{ backgroundColor: palette.card, borderColor: palette.border }}
              className="rounded-3xl p-5 border shadow-sm space-y-4"
            >
              <h3 className="font-bold text-base">Signed-In Profile</h3>
              <div
                style={{ backgroundColor: palette.bg, borderColor: palette.border }}
                className="p-4 rounded-2xl border flex items-center justify-between"
              >
                <div>
                  <p className="font-bold text-sm">{accountProfile.name}</p>
                  <p className="text-xs mt-0.5" style={{ color: palette.textMuted }}>
                    Verified via {accountProfile.method} ({accountProfile.identifier})
                  </p>
                </div>
                <button
                  onClick={() => setAccountProfile(null)}
                  style={{ backgroundColor: palette.card, borderColor: palette.border }}
                  className="px-3.5 py-2 rounded-xl border text-xs font-semibold flex items-center gap-1.5 hover:opacity-80"
                >
                  <LogOut className="w-3.5 h-3.5" />
                  Sign Out
                </button>
              </div>
            </div>
          </div>
        )}
      </main>

      {/* ======================================================================
          MODAL: ADD CUSTOM TASK / REMINDER
         ====================================================================== */}
      {showAddTaskModal && (
        <div className="fixed inset-0 z-50 bg-black/50 backdrop-blur-xs flex items-center justify-center p-4">
          <div
            style={{ backgroundColor: palette.card, borderColor: palette.border, color: palette.text }}
            className="max-w-md w-full rounded-3xl p-6 border shadow-2xl space-y-4"
          >
            <div className="flex items-center justify-between">
              <h3 className="font-bold text-lg">Add Daily Stuff or Reminder</h3>
              <button onClick={() => setShowAddTaskModal(false)} className="p-1">
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleAddCustomTask} className="space-y-3.5">
              <div>
                <label className="block text-xs font-semibold mb-1" style={{ color: palette.textMuted }}>
                  Task / Habit Name
                </label>
                <input
                  type="text"
                  required
                  value={newTaskTitle}
                  onChange={(e) => setNewTaskTitle(e.target.value)}
                  placeholder="e.g., Tahajjud Prayer, Vitamin D, Apply Face Mask"
                  style={{ backgroundColor: palette.bg, borderColor: palette.border, color: palette.text }}
                  className="w-full px-3.5 py-2.5 rounded-xl border text-sm focus:outline-none"
                />
              </div>

              <div>
                <label className="block text-xs font-semibold mb-1" style={{ color: palette.textMuted }}>
                  Note / Gentle Details
                </label>
                <input
                  type="text"
                  value={newTaskSubtitle}
                  onChange={(e) => setNewTaskSubtitle(e.target.value)}
                  placeholder="e.g., 15 minutes before sleep"
                  style={{ backgroundColor: palette.bg, borderColor: palette.border, color: palette.text }}
                  className="w-full px-3.5 py-2.5 rounded-xl border text-sm focus:outline-none"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-semibold mb-1" style={{ color: palette.textMuted }}>
                    Category
                  </label>
                  <select
                    value={newTaskCategory}
                    onChange={(e) => setNewTaskCategory(e.target.value)}
                    style={{ backgroundColor: palette.bg, borderColor: palette.border, color: palette.text }}
                    className="w-full px-3 py-2.5 rounded-xl border text-sm focus:outline-none"
                  >
                    {Object.entries(CATEGORIES).map(([k, v]) => (
                      <option key={k} value={k}>
                        {v.label}
                      </option>
                    ))}
                  </select>
                </div>

                <div>
                  <label className="block text-xs font-semibold mb-1" style={{ color: palette.textMuted }}>
                    Reminder Time
                  </label>
                  <input
                    type="time"
                    value={newTaskTime}
                    onChange={(e) => setNewTaskTime(e.target.value)}
                    style={{ backgroundColor: palette.bg, borderColor: palette.border, color: palette.text }}
                    className="w-full px-3 py-2.5 rounded-xl border text-sm focus:outline-none"
                  />
                </div>
              </div>

              <div className="grid grid-cols-2 gap-2 pt-1">
                <button
                  type="button"
                  onClick={() => setNewTaskRepeat('daily')}
                  style={
                    newTaskRepeat === 'daily'
                      ? { backgroundColor: palette.buttonBg, color: palette.buttonText }
                      : { backgroundColor: palette.bg, color: palette.textMuted, borderColor: palette.border }
                  }
                  className="py-2 rounded-xl border text-xs font-bold"
                >
                  Every Day Routine
                </button>
                <button
                  type="button"
                  onClick={() => setNewTaskRepeat('date')}
                  style={
                    newTaskRepeat === 'date'
                      ? { backgroundColor: palette.buttonBg, color: palette.buttonText }
                      : { backgroundColor: palette.bg, color: palette.textMuted, borderColor: palette.border }
                  }
                  className="py-2 rounded-xl border text-xs font-bold"
                >
                  Only on {selectedDateKey}
                </button>
              </div>

              <label
                style={{ backgroundColor: palette.bgSecondary, borderColor: palette.border }}
                className="flex items-center gap-2.5 p-3 rounded-xl border text-xs cursor-pointer"
              >
                <input
                  type="checkbox"
                  checked={newTaskAppleSync}
                  onChange={(e) => setNewTaskAppleSync(e.target.checked)}
                  className="rounded"
                />
                <span>Send push notification alert to Apple Calendar (.ics)</span>
              </label>

              <div className="flex items-center gap-2 pt-2">
                <button
                  type="button"
                  onClick={() => setShowAddTaskModal(false)}
                  style={{ backgroundColor: palette.bg, borderColor: palette.border }}
                  className="flex-1 py-3 rounded-xl border text-xs font-semibold"
                >
                  Cancel
                </button>
                <button
                  type="submit"
                  style={{ backgroundColor: palette.buttonBg, color: palette.buttonText }}
                  className="flex-1 py-3 rounded-xl text-xs font-bold shadow-md"
                >
                  Save Reminder
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* ======================================================================
          BOTTOM NAVIGATION BAR
         ====================================================================== */}
      <nav
        style={{ backgroundColor: palette.card, borderColor: palette.border }}
        className="fixed bottom-0 left-0 right-0 z-40 border-t px-4 py-2 backdrop-blur-md bg-opacity-95"
      >
        <div className="max-w-md mx-auto grid grid-cols-4 gap-1">
          <button
            onClick={() => setActiveTab('routine')}
            style={
              activeTab === 'routine'
                ? { backgroundColor: palette.accentSoft, color: palette.text }
                : { color: palette.textMuted }
            }
            className="flex flex-col items-center py-2 rounded-2xl transition"
          >
            <CheckCircle2 className="w-5 h-5" />
            <span className="text-[11px] font-bold mt-1">Daily Routine</span>
          </button>

          <button
            onClick={() => setActiveTab('analytics')}
            style={
              activeTab === 'analytics'
                ? { backgroundColor: palette.accentSoft, color: palette.text }
                : { color: palette.textMuted }
            }
            className="flex flex-col items-center py-2 rounded-2xl transition"
          >
            <BarChart3 className="w-5 h-5" />
            <span className="text-[11px] font-bold mt-1">How's My Routine</span>
          </button>

          <button
            onClick={() => setActiveTab('calendar')}
            style={
              activeTab === 'calendar'
                ? { backgroundColor: palette.accentSoft, color: palette.text }
                : { color: palette.textMuted }
            }
            className="flex flex-col items-center py-2 rounded-2xl transition"
          >
            <CalendarIcon className="w-5 h-5" />
            <span className="text-[11px] font-bold mt-1">Calendar</span>
          </button>

          <button
            onClick={() => setActiveTab('settings')}
            style={
              activeTab === 'settings'
                ? { backgroundColor: palette.accentSoft, color: palette.text }
                : { color: palette.textMuted }
            }
            className="flex flex-col items-center py-2 rounded-2xl transition"
          >
            <Palette className="w-5 h-5" />
            <span className="text-[11px] font-bold mt-1">Themes</span>
          </button>
        </div>
      </nav>
    </div>
  );
}
